(defun etaf-org-toggle-split ()
  "在单窗全宽阅读与左源码右预览之间切换。"
  (interactive)
  (let ((session (etaf-org--current-session)))
    (unless session
      (user-error "当前缓冲区不是 ETAF Org 预览"))
    (if (etaf-org--split-active-p session)
        (etaf-org--collapse-split session)
      (etaf-org--expand-split session))))
#+end_src

  (let ((palette (etaf-org--resolve-palette theme-or-palette)))
